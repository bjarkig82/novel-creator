import hashlib,json,subprocess,sys,os
from pathlib import Path
paths=['src/novel_creator/storage.py','src/novel_creator/validation.py','tests/performance/benchmark_records.py','tests/integration/test_storage.py','tests/helpers.py','tests/unit/test_bundle_metadata.py']
def hashes(): return {p:hashlib.sha256(Path(p).read_bytes()).hexdigest() for p in paths}
before=hashes()
manifest={'before':before,'environment':{name:os.environ.get(name) for name in ('PYTHONUTF8','PYTHONPATH','PYTHONDONTWRITEBYTECODE')},'commands':[]}
base=Path('tests/evaluations')
with (base/'windows-four-history-readers-full-tests-20261022.txt').open('w',encoding='utf-8') as log:
    command=[sys.executable,'-B','-m','unittest','discover','-v']
    result=subprocess.run(command,stdout=log,stderr=subprocess.STDOUT)
manifest['commands'].append({'argv':command,'exit':result.returncode})
manifest['after']=hashes(); manifest['unchanged']=before==manifest['after']
(base/'windows-four-history-readers-verification-20261022.json').write_text(json.dumps(manifest,indent=2)+'\n',encoding='utf-8')
assert manifest['unchanged']
raise SystemExit(result.returncode)
