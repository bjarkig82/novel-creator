    @contextmanager
    def _balanced_save_policy(self, enabled=True, force_large=True):
        from contextlib import ExitStack
        with ExitStack() as stack:
            for name, value in (('_SAVE_HISTORY_WORKERS', 2 if enabled else 1),
                                ('_INITIAL_HASH_WORKERS', 2 if enabled else 1),
                                ('_SAVE_HASH_WORKERS', 4 if enabled else 1)):
                stack.enter_context(patch.object(storage, name, value))
            if force_large:
                stack.enter_context(patch.object(storage, '_HISTORY_PARALLEL_MINIMUM', 1))
                stack.enter_context(patch.object(storage, '_PARALLEL_HASH_MINIMUM', 1))
            yield

    def test_balanced_save_selects_bounded_readers_and_keeps_small_bootstrap_retry_serial(self):
        self.assertEqual(storage._SAVE_HISTORY_WORKERS, 2 if os.name == 'nt' else 1)
        self.assertEqual(storage._INITIAL_HASH_WORKERS, 2 if os.name == 'nt' else 1)
        for fmt in ('sha1', 'sha256'):
            for enabled in (False, True):
                with self.subTest(fmt=fmt, enabled=enabled), self._partition_key(), repository(object_format=fmt) as root:
                    history, _, _ = self._partition_fixture(root)
                    chosen, guards = [], []
                    batch, guard = storage._batch_blobs, storage.check_managed
                    def blobs(*args, **kwargs):
                        chosen.append(kwargs.get('workers', 1))
                        return batch(*args, **kwargs)
                    def checked(*args, **kwargs):
                        guards.append(kwargs.get('hash_workers', 1))
                        return guard(*args, **kwargs)
                    with self._balanced_save_policy(enabled), patch.object(storage, '_READ_HISTORY_WORKERS', 4 if enabled else 1), patch.object(storage, '_batch_blobs', blobs), patch.object(storage, 'check_managed', checked):
                        self.assertEqual(storage.read_snapshot(root)[0], history)
                        saved = storage.save(root, envelope(root, previous=history[-1]))
                    self.assertFalse(saved['data']['cleanup_required'])
                    self.assertEqual(chosen, [4 if enabled else 1, 2 if enabled else 1])
                    self.assertEqual(guards, [2 if enabled else 1, 4 if enabled else 1])
            with self.subTest(fmt=fmt, small=True), repository(object_format=fmt, unborn=True) as root:
                first = envelope(root)
                real_pool = storage.ThreadPoolExecutor
                pools = []
                def serial_pool(*args, **kwargs):
                    count = kwargs.get('max_workers', args[0] if args else None)
                    pools.append(count)
                    self.assertEqual(count, 1, 'Small history/guards started a parallel pool')
                    return real_pool(*args, **kwargs)
                with self._balanced_save_policy(force_large=False), patch.object(storage, 'ThreadPoolExecutor', serial_pool):
                    saved = storage.save(root, first)
                    again = storage.save(root, first)
                    self.assertEqual(storage.read_history(root), [first])
                self.assertTrue(again['data']['already_saved'])
                self.assertEqual(saved['checkpoint'], again['checkpoint'])
                self.assertEqual(pools, [1, 1])

    def test_balanced_save_preserves_damaged_prefix_over_nested_worker_failures(self):
        for fmt in ('sha1', 'sha256'):
            for failure in ('launch', 'nonzero'):
                with self.subTest(fmt=fmt, failure=failure), self._partition_key(), repository(object_format=fmt) as root:
                    history, _, _ = self._partition_fixture(root)
                    (root / storage.bundle_path(history[1])).write_bytes(b'{damaged history}\n')
                    git(root, 'add', 'story/exchanges')
                    git(root, 'commit', '-m', 'Damaged second exchange')
                    commit = storage.head(root)
                    entries = storage.committed_entries(root, commit)
                    expected = self._partition_result(root, commit, entries, 1)
                    self.assertEqual(expected[0:2], ('error', 4))
                    self.assertEqual(expected[3]['sequence'], 1)
                    pending = envelope(root, previous=history[-1])
                    index = (root / '.git/index').read_bytes()
                    late = entries[sorted(p for p in entries if p.startswith('story/exchanges/'))[4]][1]
                    original, guard = storage.git, storage.check_managed
                    guard_done = threading.Event()
                    history_done, hash_done = [], []
                    def checked(*args, **kwargs):
                        try:
                            return guard(*args, **kwargs)
                        finally:
                            guard_done.set()
                    def flaky(root, *args, **kwargs):
                        if args[0] == 'hash-object' and '--stdin-paths' in args:
                            hash_done.append(kwargs['input'])
                            return b''
                        if args[0] != 'cat-file':
                            return original(root, *args, **kwargs)
                        requested = kwargs['input'].decode().splitlines()
                        if len(requested) == 8:
                            self.assertEqual(len(history_done), 2, 'History fallback did not join both children')
                            self.assertTrue(guard_done.is_set())
                            return original(root, *args, **kwargs)
                        try:
                            if requested[0] == late:
                                self.assertTrue(guard_done.wait(10), 'First guard did not finish')
                                if failure == 'launch':
                                    raise OSError('Injected history worker launch failure')
                                return subprocess.CompletedProcess(args, 1, b'', b'')
                            return original(root, *args, **kwargs)
                        finally:
                            history_done.append(requested[0])
                    with self._balanced_save_policy(), patch.object(storage, 'git', flaky), patch.object(storage, 'check_managed', checked):
                        with self.assertRaises(ToolError) as error:
                            storage.save(root, pending)
                    self.assertEqual((error.exception.code, error.exception.affected_refs, error.exception.checkpoint), expected[1:])
                    self.assertEqual(len(hash_done), 2)
                    self.assertTrue(guard_done.is_set())
                    self.assertEqual(storage.head(root), commit)
                    self.assertEqual((root / '.git/index').read_bytes(), index)
                    self.assertFalse((root / storage.bundle_path(pending)).exists())
                    self.assertFalse((root / '.novel-creator/local' / (pending['exchange_id'] + '.json')).exists())
                    self.assertFalse((root / '.novel-creator/local/write.lock').exists())

    def test_balanced_first_guard_failures_join_before_unlock_without_pending_work(self):
        for fmt in ('sha1', 'sha256'):
            for failure in ('launch', 'process', 'count', 'thread'):
                with self.subTest(fmt=fmt, failure=failure), self._partition_key(), repository(object_format=fmt) as root:
                    history, commit, _ = self._partition_fixture(root)
                    pending = envelope(root, previous=history[-1])
                    index = (root / '.git/index').read_bytes()
                    original, real_pool = storage.git, storage.ThreadPoolExecutor
                    started, finished = [], []
                    class FailSecondHashSubmit(real_pool):
                        def submit(self, fn, *args, **kwargs):
                            if fn.__name__ == 'chunk_hashes':
                                if getattr(self, 'hash_submitted', False):
                                    raise RuntimeError('Injected nested hash thread-start failure')
                                self.hash_submitted = True
                            return super().submit(fn, *args, **kwargs)
                    def flaky(root, *args, **kwargs):
                        if args[0] != 'hash-object' or '--stdin-paths' not in args:
                            return original(root, *args, **kwargs)
                        started.append(kwargs['input'])
                        self.assertTrue((root / '.novel-creator/local/write.lock').exists())
                        try:
                            if failure == 'launch':
                                raise OSError('Injected first-guard launch failure')
                            if failure == 'process':
                                raise ToolError(5, 'Injected first-guard process failure')
                            if failure == 'count':
                                return b''
                            return original(root, *args, **kwargs)
                        finally:
                            self.assertTrue((root / '.novel-creator/local/write.lock').exists())
                            finished.append(kwargs['input'])
                    with self._balanced_save_policy(), patch.object(storage, 'git', flaky), patch.object(storage, 'ThreadPoolExecutor', FailSecondHashSubmit if failure == 'thread' else real_pool):
                        with self.assertRaises(ToolError) as error:
                            storage.save(root, pending)
                    self.assertEqual(error.exception.code, 5)
                    self.assertEqual(error.exception.checkpoint['sequence'], 8)
                    self.assertEqual(len(started), 1 if failure == 'thread' else 2)
                    self.assertCountEqual(finished, started)
                    self.assertEqual(storage.head(root), commit)
                    self.assertEqual((root / '.git/index').read_bytes(), index)
                    self.assertFalse((root / storage.bundle_path(pending)).exists())
                    self.assertFalse((root / '.novel-creator/local' / (pending['exchange_id'] + '.json')).exists())
                    self.assertFalse((root / '.novel-creator/local/write.lock').exists())

    def test_balanced_first_guard_hashes_real_root_after_metadata_despite_same_stat_decoy(self):
        for fmt in ('sha1', 'sha256'):
            with self.subTest(fmt=fmt), self._partition_key(), repository(object_format=fmt) as root:
                history, _, _ = self._partition_fixture(root)
                target = root / '.novel-creator/src/guard.txt'
                target.parent.mkdir(parents=True, exist_ok=True)
                target.write_bytes(b'original\n')
                git(root, 'add', '.novel-creator/src')
                git(root, 'commit', '-m', 'First-guard byte resource')
                commit = storage.head(root)
                entries = storage.committed_entries(root, commit)
                index = (root / '.git/index').read_bytes()
                pending = envelope(root, previous=history[-1])
                before = target.stat()
                original_hash = storage._hash_managed
                changed = threading.Event()
                with tempfile.TemporaryDirectory(prefix='novel-balanced-decoy-') as directory:
                    decoy = Path(directory)
                    for path in entries:
                        destination = decoy / path
                        destination.parent.mkdir(parents=True, exist_ok=True)
                        destination.write_bytes((root / path).read_bytes())
                    def redirected(root, managed, workers):
                        self.assertEqual(workers, 2)
                        self.assertFalse(changed.is_set())
                        changed.set()
                        target.write_bytes(b'tampered\n')
                        os.utime(target, ns=(before.st_atime_ns, before.st_mtime_ns))
                        git(root, 'config', 'core.worktree', str(decoy))
                        return original_hash(root, managed, workers)
                    try:
                        with self._balanced_save_policy(), patch.object(storage, '_hash_managed', redirected):
                            with self.assertRaises(ToolError) as error:
                                storage.save(root, pending)
                        self.assertEqual(error.exception.code, 3)
                        self.assertEqual(error.exception.checkpoint['sequence'], 8)
                        self.assertEqual(target.read_bytes(), b'tampered\n')
                        self.assertEqual(storage.head(root), commit)
                        self.assertEqual((root / '.git/index').read_bytes(), index)
                        self.assertFalse((root / '.novel-creator/local/write.lock').exists())
                        self.assertFalse((root / '.novel-creator/local' / (pending['exchange_id'] + '.json')).exists())
                    finally:
                        git(root, 'config', '--unset', 'core.worktree')

    def test_balanced_edit_after_first_guard_is_caught_then_unchanged_retry_publishes_once(self):
        for fmt in ('sha1', 'sha256'):
            with self.subTest(fmt=fmt), self._partition_key(), repository(object_format=fmt) as root:
                history, _, _ = self._partition_fixture(root)
                target = root / '.novel-creator/src/guard.txt'
                target.parent.mkdir(parents=True, exist_ok=True)
                target.write_bytes(b'original\n')
                git(root, 'add', '.novel-creator/src')
                git(root, 'commit', '-m', 'Post-first-guard byte resource')
                commit = storage.head(root)
                index = (root / '.git/index').read_bytes()
                pending = envelope(root, previous=history[-1])
                before = target.stat()
                guard, blobs = storage.check_managed, storage._batch_blobs
                changed = threading.Event()
                def checked(*args, **kwargs):
                    result = guard(*args, **kwargs)
                    if kwargs.get('hash_workers') == 2:
                        target.write_bytes(b'tampered\n')
                        os.utime(target, ns=(before.st_atime_ns, before.st_mtime_ns))
                        changed.set()
                    return result
                def wait_for_edit(*args, **kwargs):
                    for number, item in enumerate(blobs(*args, **kwargs)):
                        if number == 0:
                            self.assertTrue(changed.wait(10), 'First guard/edit did not finish before validation')
                        yield item
                with self._balanced_save_policy(), patch.object(storage, 'check_managed', checked), patch.object(storage, '_batch_blobs', wait_for_edit):
                    with self.assertRaises(ToolError) as error:
                        storage.save(root, pending)
                self.assertEqual(error.exception.code, 3)
                self.assertTrue(changed.is_set())
                self.assertEqual(storage.head(root), commit)
                self.assertEqual((root / '.git/index').read_bytes(), index)
                self.assertEqual(target.read_bytes(), b'tampered\n')
                self.assertFalse((root / '.novel-creator/local/write.lock').exists())
                receipt = storage.read_receipt(root, root / '.novel-creator/local' / (pending['exchange_id'] + '.json'))
                self.assertEqual(receipt['state'], 'pending')
                self.assertIn('candidate', receipt)
                target.write_bytes(b'original\n')
                with self._balanced_save_policy():
                    saved = storage.save(root, pending)
                    again = storage.save(root, pending)
                self.assertFalse(saved['data']['cleanup_required'])
                self.assertTrue(again['data']['already_saved'])
                self.assertEqual(saved['checkpoint'], again['checkpoint'])
                self.assertEqual(git(root, 'rev-list', '--count', commit + '..HEAD'), '1')
                self.assertEqual(storage.read_history(root), history + [pending])
                self.assertEqual(git(root, 'diff', '--cached', '--exit-code'), '')

    @contextmanager
    def _partition_key(self):
        from tests.performance import benchmark_records as benchmark
        with benchmark.benchmark_key(True):
            yield

    def _partition_fixture(self, root):
        from tests.performance import benchmark_records as benchmark
        novel_id = records.loads((root / '.novel-creator/manifest.json').read_bytes())['novel_id']
        history, _ = benchmark.generate(novel_id, exchanges=8, sessions=2, assertions=16, minimum_bytes=0)
        for index, bundle in enumerate(history):
            target = root / storage.bundle_path(bundle)
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(benchmark.serialized(bundle))
            if index == 0:
                git(root, 'add', 'story/exchanges')
                git(root, 'commit', '-m', 'Validated one-exchange prefix')
        git(root, 'add', 'story/exchanges')
        git(root, 'commit', '-m', 'Complete candidate fault fixture')
        storage.read_history(root)
        return (history, storage.head(root), storage.committed_entries(root, storage.head(root)))

    def _partition_result(self, root, commit, entries, workers):
        with patch.object(storage, '_HISTORY_PARALLEL_MINIMUM', 1):
            try:
                history, chain = storage._read_snapshot(root, commit, entries, history_workers=workers)
                return ('ok', history, chain._state)
            except ToolError as error:
                return ('error', error.code, error.affected_refs, error.checkpoint)

    def test_partitioned_history_preserves_binary_framing_and_nonzero_outputs(self):
        for fmt in ('sha1', 'sha256'):
            with self.subTest(fmt=fmt), self._partition_key(), repository(object_format=fmt) as root:
                _, commit, entries = self._partition_fixture(root)
                paths = sorted((p for p in entries if p.startswith('story/exchanges/')))
                raw = {entries[p][1]: (root / p).read_bytes() for p in paths}
                oids = [entries[p][1] for p in paths]
                original = storage.git
                cases = [(kind, index) for kind in ('missing', 'header', 'truncated') for index in (1, 2, 3, 4, 5, 6)]
                cases += [('trailing', index) for index in (1, 3, 5)]
                cases += [('nonzero_partial', index) for index in (0, 1, 2, 3, 4, 5)] + [('nonzero_complete', 0)]
                for kind, index in cases:
                    with self.subTest(fmt=fmt, kind=kind, index=index):

                        def fake(root, *args, **kwargs):
                            if args[0] != 'cat-file':
                                return original(root, *args, **kwargs)
                            requested = kwargs['input'].decode().splitlines()
                            data = []
                            status = 0
                            for oid in requested:
                                body = raw[oid]
                                header = f'{oid} blob {len(body)}\n'.encode()
                                frame = header + body + b'\n'
                                if oid == oids[index]:
                                    if kind == 'missing':
                                        frame = f'{oid} missing\n'.encode()
                                    if kind == 'header':
                                        frame = b'invalid header\n' + body + b'\n'
                                    if kind == 'truncated':
                                        data.append(header + body[:-1])
                                        break
                                    if kind == 'trailing':
                                        frame += b'unexpected trailing bytes\n'
                                data.append(frame)
                                if kind == 'nonzero_partial' and oid == oids[index]:
                                    status = 1
                                    break
                            if kind == 'nonzero_complete':
                                status = 1
                            return subprocess.CompletedProcess(args, status, b''.join(data), b'')
                        with patch.object(storage, 'git', fake):
                            expected = self._partition_result(root, commit, entries, 1)
                            for workers in (2, 4):
                                with self.subTest(workers=workers):
                                    actual = self._partition_result(root, commit, entries, workers)
                                    self.assertEqual(actual, expected)
                                    self.assertEqual(actual[0], 'ok' if kind == 'nonzero_complete' else 'error')

    def test_partitioned_history_preserves_earlier_damage_and_guard_precedence(self):
        for fmt in ('sha1', 'sha256'):
            with self.subTest(fmt=fmt), self._partition_key(), repository(object_format=fmt) as root:
                history, _, _ = self._partition_fixture(root)
                damaged = root / storage.bundle_path(history[1])
                damaged.write_bytes(b'{bad history}\n')
                git(root, 'add', 'story/exchanges')
                git(root, 'commit', '-m', 'Malformed second exchange')
                commit = storage.head(root)
                entries = storage.committed_entries(root, commit)
                expected = self._partition_result(root, commit, entries, 1)
                self.assertEqual(expected[0:2], ('error', 4))
                self.assertEqual(expected[3]['sequence'], 1)
                paths = sorted((p for p in entries if p.startswith('story/exchanges/')))
                late = entries[paths[4]][1]
                original = storage.git
                completed = []
                ready = threading.Event()

                def flaky(root, *args, **kwargs):
                    if args[0] != 'cat-file':
                        return original(root, *args, **kwargs)
                    requested = kwargs['input'].decode().splitlines()
                    if len(requested) == 8:
                        self.assertEqual(len(completed), 4, 'Fallback began before every worker joined')
                        return original(root, *args, **kwargs)
                    try:
                        if requested[0] == late:
                            self.assertTrue(ready.wait(10), 'Background guard did not fail')
                            raise OSError('Injected late chunk launch failure')
                        return original(root, *args, **kwargs)
                    finally:
                        completed.append(requested[0])
                ready.set()
                with patch.object(storage, 'git', flaky):
                    self.assertEqual(self._partition_result(root, commit, entries, 4), expected)
                ready.clear()
                completed.clear()
                working = root / storage.bundle_path(history[0])
                working.write_bytes(working.read_bytes() + b' ')
                index = (root / '.git/index').read_bytes()
                original_guard = storage.check_managed

                def guard(*args, **kwargs):
                    try:
                        return original_guard(*args, **kwargs)
                    except ToolError as error:
                        self.assertEqual(error.code, 3)
                        ready.set()
                        raise
                original_reader = storage._read_snapshot

                def force_parallel(*args, **kwargs):
                    kwargs['history_workers'] = 4
                    return original_reader(*args, **kwargs)
                with patch.object(storage, '_HISTORY_PARALLEL_MINIMUM', 1), patch.object(storage, 'git', flaky), patch.object(storage, 'check_managed', guard), patch.object(storage, '_read_snapshot', force_parallel):
                    with self.assertRaises(ToolError) as error:
                        storage.save(root, envelope(root, previous=history[-1]))
                self.assertEqual((error.exception.code, error.exception.affected_refs, error.exception.checkpoint), expected[1:])
                self.assertEqual(storage.head(root), commit)
                self.assertEqual((root / '.git/index').read_bytes(), index)
                self.assertFalse((root / '.novel-creator/local/write.lock').exists())

    def test_partitioned_history_joins_partial_submission_before_single_fallback(self):
        for fmt in ('sha1', 'sha256'):
            with self.subTest(fmt=fmt), self._partition_key(), repository(object_format=fmt) as root:
                history, commit, entries = self._partition_fixture(root)
                original = storage.git
                done = threading.Event()
                calls = []
                real_pool = storage.ThreadPoolExecutor

                class FailSecondSubmit(real_pool):

                    def submit(self, *args, **kwargs):
                        if getattr(self, 'submitted', False):
                            raise RuntimeError('Injected worker start failure')
                        self.submitted = True
                        return super().submit(*args, **kwargs)

                def traced(root, *args, **kwargs):
                    if args[0] != 'cat-file':
                        return original(root, *args, **kwargs)
                    requested = kwargs['input'].decode().splitlines()
                    calls.append(len(requested))
                    if len(requested) == 8:
                        self.assertTrue(done.is_set(), 'Started child was not joined before fallback')
                        return original(root, *args, **kwargs)
                    try:
                        return original(root, *args, **kwargs)
                    finally:
                        done.set()
                with patch.object(storage, 'ThreadPoolExecutor', FailSecondSubmit), patch.object(storage, 'git', traced):
                    result = self._partition_result(root, commit, entries, 4)
                self.assertEqual(result[0:2], ('ok', history))
                self.assertEqual(calls, [2, 8])

    def test_parallel_hash_guard_checks_every_path_after_all_metadata(self):
        for object_format in ('sha1', 'sha256'):
            with self.subTest(object_format=object_format), repository(object_format=object_format,
                    prefix='novel Þóra with spaces-') as root:
                for number in range(7):
                    target = root / f'.novel-creator/src/Þóra {number}.py'
                    target.parent.mkdir(parents=True, exist_ok=True)
                    target.write_bytes(b'original managed data\n')
                git(root, 'add', '--', '.novel-creator/src')
                git(root, 'commit', '-m', 'Disposable complete guard byte fixture')
                initial = storage.head(root)
                entries = storage.committed_entries(root, initial)
                expected = [path for path in entries if path.startswith('.novel-creator/')]
                checked, hashed = [], []
                original_git, original_check = storage.git, storage._checked_path
                def checked_path(root, path, **kwargs):
                    result = original_check(root, path, **kwargs)
                    checked.append(path)
                    return result
                def traced(root, *args, **kwargs):
                    if args[0] == 'hash-object':
                        self.assertEqual(checked, expected)
                        self.assertEqual(args, ('hash-object', '--no-filters', '--stdin-paths'))

                        hashed.extend(json.loads(path) for path in kwargs['input'].decode().splitlines())
                    return original_git(root, *args, **kwargs)
                with patch.object(storage, '_PARALLEL_HASH_MINIMUM', 1), \
                        patch.object(storage, '_checked_path', checked_path), patch.object(storage, 'git', traced):
                    self.assertEqual(storage.check_managed(root, initial, [], entries=entries, hash_workers=4), entries)

    def test_parallel_final_guard_failures_join_workers_before_releasing_writer(self):
        for object_format in ('sha1', 'sha256'):
            for failure in ('incomplete', 'process', 'launch', 'thread'):
                with self.subTest(object_format=object_format, failure=failure), repository(
                        object_format=object_format) as root:
                    first = envelope(root)
                    storage.save(root, first)
                    for number in range(6):
                        target = root / f'.novel-creator/src/resource {number}.py'
                        target.parent.mkdir(parents=True, exist_ok=True)
                        target.write_bytes(b'original managed data\n')
                    git(root, 'add', '--', '.novel-creator/src')
                    git(root, 'commit', '-m', 'Disposable save resources')
                    initial = storage.head(root)
                    index = (root / '.git/index').read_bytes()
                    second = envelope(root, previous=first)
                    original, original_guard = storage.git, storage.check_managed
                    original_executor = storage.ThreadPoolExecutor
                    phase, completed, selected = threading.Event(), [], []
                    class UnavailableExecutor(original_executor):
                        def submit(self, *args, **kwargs):
                            raise RuntimeError("Injected failure to start a worker thread")
                    def executor(*args, **kwargs):
                        selected_executor = UnavailableExecutor if failure == 'thread' and phase.is_set() else original_executor
                        return selected_executor(*args, **kwargs)
                    def guard(*args, **kwargs):
                        selected.append(kwargs.get('hash_workers', 1))
                        return original_guard(*args, **kwargs)
                    def fault(point):
                        if point == 'before_ref':
                            phase.set()
                    def fail_hash(root, *args, **kwargs):
                        if args[0] != 'hash-object' or '--stdin-paths' not in args or not phase.is_set():
                            return original(root, *args, **kwargs)
                        paths = [json.loads(path) for path in kwargs['input'].decode().splitlines()]
                        self.assertTrue((root / '.novel-creator/local/write.lock').exists())
                        try:
                            if paths[0] == '.novel-creator/manifest.json':
                                # This conflict must not mask another chunk's
                                # incomplete output or Git process failure.
                                return (('0' * len(initial) + '\n') * len(paths)).encode()
                            if failure == 'launch':
                                raise OSError('Injected Git process launch failure')
                            if failure == 'process':
                                raise ToolError(5, 'Injected Git process failure')
                            return b''
                        finally:
                            completed.extend(paths)
                    with patch.object(storage, '_PARALLEL_HASH_MINIMUM', 1), \
                            patch.object(storage, '_SAVE_HASH_WORKERS', 4), \
                            patch.object(storage, 'ThreadPoolExecutor', executor), \
                            patch.object(storage, 'check_managed', guard), patch.object(storage, 'git', fail_hash):
                        with self.assertRaises(ToolError) as error:
                            storage.save(root, second, fault=fault)
                    self.assertEqual(error.exception.code, 5)
                    self.assertEqual(selected, [storage._INITIAL_HASH_WORKERS, 4])
                    expected = [path for path in storage.committed_entries(root, initial)
                                if path.startswith(('.novel-creator/', 'story/exchanges/'))]
                    self.assertCountEqual(completed, [] if failure == 'thread' else expected)
                    self.assertEqual(storage.head(root), initial)
                    self.assertEqual((root / '.git/index').read_bytes(), index)
                    self.assertFalse((root / '.novel-creator/local/write.lock').exists())
                    receipt = storage.read_receipt(root, root / '.novel-creator/local' / (second['exchange_id'] + '.json'))
                    self.assertEqual(receipt['state'], 'pending')
                    self.assertIn('candidate', receipt)

    def test_held_lock_reports_failure_context_without_mutating_owner_or_saved_work(self):

            if context == 'non_utf8' and os.name == 'nt':
                continue  # Windows filenames do not expose arbitrary byte names.
            with self.subTest(context=context), repository(unborn=context == 'unborn') as root:
                expected = None
                if context in ('saved', 'non_utf8'):
                    first = envelope(root)

                    data = envelope(root)

                    (root / relative).write_text('[]')

                        stream.write(b'Unrelated user file')

                git(root, 'add', 'notes.txt')
