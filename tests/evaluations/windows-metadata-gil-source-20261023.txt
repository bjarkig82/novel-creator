"""Diagnostic only: fresh Win32 queries with/without Python GIL handoffs, x64."""
import argparse
import ctypes
from ctypes import wintypes
import hashlib
import json
from pathlib import Path
import statistics
import subprocess
import sys
import time
from unittest.mock import patch

sys.path.insert(0, str(Path.cwd()))
from novel_creator import records, storage
from novel_creator.context import read
from tests.helpers import repository, git
from tests.performance import benchmark_records as benchmark

parser = argparse.ArgumentParser()
parser.add_argument('--root', type=Path)
parser.add_argument('--hold-gil', type=int, default=0)
args = parser.parse_args()
assert ctypes.sizeof(ctypes.c_void_p) == 8, 'Diagnostic uses the unified Windows x64 calling convention'

if args.root:
    prior = storage.read_history(args.root)
    exchange = records.new_exchange(prior[0]['novel_id'], previous=prior[-1], expected_head=storage.head(args.root),
        messages=[{'message_id': benchmark.uid(), 'role': 'user', 'text': 'Continue the benchmark.'},
                  {'message_id': benchmark.uid(), 'role': 'assistant', 'text': 'Benchmark continuation.'}])
    exchange['payload_digest'] = records.digest(exchange)
    if args.hold_gil:
        library = ctypes.PyDLL('kernel32', use_last_error=True)
        api = library.GetFileAttributesW
        api.argtypes = [wintypes.LPCWSTR]
        api.restype = wintypes.DWORD
    else:
        api = storage._windows_file_attributes()
    assert api is not None
    calls, hashes = [], []
    original_git = storage.git
    def query(path):
        calls.append(path)
        return api(path)
    def counted_git(root, *a, **kwargs):
        if a[0] == 'hash-object' and '--stdin-paths' in a:
            assert kwargs['env'] == {'GIT_WORK_TREE': str(root)}
            hashes.append(kwargs['input'].count(b'\n'))
        return original_git(root, *a, **kwargs)
    with patch.object(storage, '_windows_file_attributes', return_value=query), patch.object(storage, 'git', counted_git):
        started = time.perf_counter()
        result = storage.save(args.root, exchange)
        elapsed = time.perf_counter() - started
    assert result['ok'] and not result['data']['cleanup_required']
    assert len(calls) == 2 * len(prior), len(calls)
    assert len(hashes) == 6 and sum(hashes) == 2 * (len(prior) + 1), hashes
    print(json.dumps({'hold_gil': args.hold_gil, 'seconds': elapsed,
        'native_queries': len(calls), 'hashed_files': sum(hashes), 'sequence': result['checkpoint']['sequence']}))
    raise SystemExit(0)

fingerprint = hashlib.sha256(Path(storage.__file__).read_bytes()).hexdigest()
with benchmark.benchmark_key(True):
    for layout in ('ordinary', 'packed'):
        with repository(prefix='novel metadata GIL ') as root:
            novel = records.loads((root / '.novel-creator/manifest.json').read_bytes())['novel_id']
            history, fixture = benchmark.generate(novel)
            for exchange in history:
                path = root / storage.bundle_path(exchange)
                path.parent.mkdir(parents=True, exist_ok=True)
                path.write_bytes(benchmark.serialized(exchange))
            git(root, 'add', 'story/exchanges')
            git(root, 'commit', '-m', 'Metadata GIL diagnostic fixture')
            if layout == 'packed':
                git(root, 'repack', '-ad')
            del history, exchange
            read(root)
            fixture['fully_validated'] = True
            print(json.dumps({'layout': layout, 'fixture': fixture, 'source_sha256': fingerprint,
                'method': 'Eight fresh complete saves ABBA/ABBA; identical fresh queries and all six hash partitions counted. Direct Git. Diagnostic, not p95.'}), flush=True)
            samples = {0: [], 1: []}
            for enabled in [0, 1, 1, 0] * 2:
                result = subprocess.run([sys.executable, '-B', __file__, '--root', str(root), '--hold-gil', str(enabled)],
                    capture_output=True, text=True, encoding='utf-8', timeout=45)
                if result.returncode:
                    raise RuntimeError(result.stderr[-3000:])
                row = json.loads(result.stdout)
                samples[enabled].append(row['seconds'])
                print(json.dumps(row), flush=True)
            print(json.dumps({'layout': layout, 'medians': {str(k): statistics.median(v) for k, v in samples.items()}}), flush=True)
assert fingerprint == hashlib.sha256(Path(storage.__file__).read_bytes()).hexdigest()
