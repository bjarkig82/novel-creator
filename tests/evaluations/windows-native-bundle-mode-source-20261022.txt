ordinary=(stat.FILE_ATTRIBUTE_NORMAL|stat.FILE_ATTRIBUTE_ARCHIVE|stat.FILE_ATTRIBUTE_READONLY|stat.FILE_ATTRIBUTE_HIDDEN|stat.FILE_ATTRIBUTE_SYSTEM)
def bundle_mode(path):
    attributes=get_attributes(path)
    # Only fresh ordinary-file metadata is used. Every other attribute/error
    # goes through the existing lstat path, including reparse points and dirs.
    if attributes and not attributes & ~ordinary:
        return stat.S_IFREG
    return os.lstat(path).st_mode
