"""Corrected candidate diagnostics in fresh workers; not a release benchmark."""
import argparse,hashlib,json,os,shutil,statistics,subprocess,sys,time
from pathlib import Path
from unittest.mock import patch
sys.path.insert(0,str(Path.cwd()))
from novel_creator import records,storage
from novel_creator.context import read
from tests.helpers import repository,git
from tests.performance import benchmark_records as benchmark
import history_partition_candidate as candidate

parser=argparse.ArgumentParser();parser.add_argument('--worker',choices=['read','save']);parser.add_argument('--root',type=Path);parser.add_argument('--workers',type=int,default=0)
args=parser.parse_args()
if args.worker:
    original=storage._batch_blobs
    def selected(root,entries,paths):return candidate.partitioned_blobs(root,entries,paths,workers=args.workers) if args.workers else original(root,entries,paths)
    if args.worker=='save':
        history=storage.read_history(args.root)
        exchange=records.new_exchange(history[0]['novel_id'],previous=history[-1],expected_head=storage.head(args.root),messages=[{'message_id':benchmark.uid(),'role':'user','text':'Continue the benchmark.'},{'message_id':benchmark.uid(),'role':'assistant','text':'Benchmark continuation.'}])
        exchange['payload_digest']=records.digest(exchange)
    events=[];original_git=storage.git
    def traced(root,*arguments,**kwargs):
        started=time.perf_counter()
        try:return original_git(root,*arguments,**kwargs)
        finally:events.append({'command':arguments[0],'seconds':time.perf_counter()-started})
    with patch.object(storage,'_batch_blobs',selected),patch.object(storage,'git',traced):
        started=time.perf_counter()
        if args.worker=='save':
            result=storage.save(args.root,exchange);count=result['checkpoint']['sequence'];assert not result['data']['cleanup_required']
        else:result=read(args.root);count=len(result['data']['records'])
        elapsed=time.perf_counter()-started
    if args.worker=='save':assert len(events)==25+max(0,args.workers-1),len(events)
    print(json.dumps({'operation':args.worker,'history_workers':args.workers,'seconds':elapsed,'observed_count':count,'git':events}))
    raise SystemExit(0)

frozen=hashlib.sha256(Path(storage.__file__).read_bytes()).hexdigest()
candidate_sha=hashlib.sha256(Path(candidate.__file__).read_bytes()).hexdigest()
with benchmark.benchmark_key(True),repository() as root:
    novel_id=records.loads((root/'.novel-creator/manifest.json').read_bytes())['novel_id']
    history,fixture=benchmark.generate(novel_id)
    for exchange in history:
        target=root/storage.bundle_path(exchange);target.parent.mkdir(parents=True,exist_ok=True);target.write_bytes(benchmark.serialized(exchange))
    git(root,'add','story/exchanges');git(root,'commit','-m','Corrected partitioned-history fresh-worker fixture')
    del history,exchange
    read(root);fixture['fully_validated']=True
    print(json.dumps({'fixture':fixture,'storage_sha256':frozen,'candidate_sha256':candidate_sha,'git_executable':shutil.which('git'),'method':'Four balanced samples/configuration in fresh subprocesses; complete context reads and complete saves; one/two/four history readers; original byte guards remain; original serial payload-preparation history retained for every save; no source changes or p95 claim'}),flush=True)
    all_samples={}
    for operation in ('read','save'):
        samples={0:[],2:[],4:[]}
        for workers in [0,2,4,4,2,0]*2:
            result=subprocess.run([sys.executable,'-B',__file__,'--worker',operation,'--root',str(root),'--workers',str(workers)],capture_output=True,text=True,encoding='utf-8',timeout=45)
            if result.returncode:raise RuntimeError('Diagnostic worker failed: '+result.stderr[-2000:])
            row=json.loads(result.stdout)
            samples[workers].append(row['seconds']);print(json.dumps(row),flush=True)
            if operation=='read':assert row['observed_count']==15101
        all_samples[operation]={str(w):statistics.median(values) for w,values in samples.items()}
    print(json.dumps({'medians':all_samples}),flush=True)
    assert frozen==hashlib.sha256(Path(storage.__file__).read_bytes()).hexdigest()
    assert candidate_sha==hashlib.sha256(Path(candidate.__file__).read_bytes()).hexdigest()
